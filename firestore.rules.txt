rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /archive1025_results/{resultId} {
      allow create: if request.auth != null
        && request.resource.data.keys().hasAll([
          'schemaVersion','sessionId','schoolYear','studentId','mode',
          'actualSeconds','hintCount','hintPenaltySeconds','adjustedSeconds',
          'room1Seconds','room2Seconds','room3Seconds','hintsUsed',
          'completedAtISO','submittedAt','anonymousUid'
        ])
        && request.resource.data.keys().hasOnly([
          'schemaVersion','sessionId','schoolYear','studentId','mode',
          'actualSeconds','hintCount','hintPenaltySeconds','adjustedSeconds',
          'room1Seconds','room2Seconds','room3Seconds','hintsUsed',
          'completedAtISO','submittedAt','anonymousUid'
        ])
        && request.resource.data.schoolYear is string
        && request.resource.data.studentId is string
        && request.resource.data.mode in ['beginner','challenge']
        && request.resource.data.actualSeconds is int
        && request.resource.data.hintCount is int
        && request.resource.data.hintPenaltySeconds is int
        && request.resource.data.adjustedSeconds is int
        && request.resource.data.hintsUsed is list;
      allow read, update, delete: if false;
    }
  }
}
